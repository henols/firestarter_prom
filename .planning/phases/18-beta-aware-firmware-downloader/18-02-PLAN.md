import logging

# Module-level helpers (before create_firmware_args):
def _validate_firmware_version(value: str) -> str:
    from firestarter.firmware import FIRMWARE_VERSION_RE
    if not FIRMWARE_VERSION_RE.match(value):
        raise argparse.ArgumentTypeError(
            f"Invalid firmware version {value!r}. "
            "Expected X.Y.Z, X.Y.ZbN, or X.Y.ZrcN (e.g. 3.1.0, 3.1.0b2, 3.1.0rc1)."
        )
    return value

def _maybe_auto_route_to_pre(args) -> None:
    """D-22 / D-25. NO logger parameter (revision warning #6) — uses logging.getLogger(__name__)
    so pytest's caplog captures by logger name automatically. TestMagicDefault must NOT pass
    a logger argument.
    """
    logger = logging.getLogger(__name__)
    if not (getattr(args, "install", False)
            and not getattr(args, "pre", False)
            and not getattr(args, "firmware_version", None)):
        return
    try:
        import firestarter as _pkg
        from packaging.version import Version, InvalidVersion
        try:
            if Version(_pkg.__version__).is_prerelease:
                args.pre = True
                logger.info(
                    "Beta app detected — defaulting to --pre. "
                    "Use --firmware-version X.Y.Z to pin a stable version."
                )
        except InvalidVersion:
            pass
    except ImportError:
        pass

def create_firmware_args(parser):
    fw_parser = parser.add_parser("fw", help="Firmware version.")
    install_group = fw_parser.add_mutually_exclusive_group()
    install_group.add_argument("-i", "--install", action="store_true", help="...")
    install_group.add_argument("--list", action="store_true", help="List available firmware releases.")
    # Channel group: --pre XOR --firmware-version XOR --stable (D-13 / D-19).
    # Revision blocker #1 CLEANEST option: --stable joins this group; argparse enforces
    # the 3-way mutex natively. In install context --stable is a redundant no-op (stable
    # is default); in --list context --stable filters by channel.
    channel_group = fw_parser.add_mutually_exclusive_group()
    channel_group.add_argument("--pre", action="store_true", help="Fetch latest pre-release firmware.")
    channel_group.add_argument("--firmware-version", type=_validate_firmware_version,
                                metavar="VERSION", help="Pin exact firmware version (e.g. 3.1.0, 3.1.0b2, 3.1.0rc1).")
    channel_group.add_argument("--stable", action="store_true",
                                help="Explicitly select stable channel. With --list, filters to stable releases only.")
    fw_parser.add_argument("-b", "--board", type=str, default="uno",
                            choices=["uno", "leonardo"], help="...")
    fw_parser.add_argument("--avrdude-path", type=str, help="...")
    fw_parser.add_argument("-c", "--avrdude-config-path", type=str, help="...")
    fw_parser.add_argument("-f", "--force", action="store_true", help="...")
    fw_parser.add_argument("--json", action="store_true",
                            help="Output --list results as JSON array (only with --list).")
    return fw_parser  # MUST return for fw_parser.error() in dispatch

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-18-04 | Tampering | `--firmware-version` injection through to URL path or shell exec | mitigate | `FIRMWARE_VERSION_RE = re.compile(r'^[0-9]+\.[0-9]+\.[0-9]+((b|rc)[0-9]+)?$')` rejects any non-conforming string at argparse `type=` time (D-07), BEFORE the value reaches `.format()` or `requests.get`. No shell exec on the value at any point.
