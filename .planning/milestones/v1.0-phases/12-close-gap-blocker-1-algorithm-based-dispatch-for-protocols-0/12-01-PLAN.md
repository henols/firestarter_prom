- if protocol == 0x10 -> configure_flash_intel
- if protocol == 0x0D -> configure_eeprom28c

- if protocol in {0x05, 0x35, 0x39} -> configure_flash4         [new — 0x39 by analogy with 0x35; PATTERNS.md note]
- if protocol in {0x07, 0x08, 0x0B} -> configure_eprom          [new]
- if protocol in {0x0E, 0x27, 0x28, 0x29} -> configure_sram     [new]
- fallback mem_type: 1->eprom, 4->sram, 3->flash3, 5->flash4
- else: "Memory type 0x%02x not supported" -> ERROR
